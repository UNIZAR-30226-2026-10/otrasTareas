package com.UNIZAR_30226_2026_10.data.local

import android.content.Context
import androidx.datastore.preferences.core.booleanPreferencesKey
import androidx.datastore.preferences.core.edit
import androidx.datastore.preferences.core.stringPreferencesKey
import androidx.datastore.preferences.preferencesDataStore
import kotlinx.coroutines.flow.first

// Extensión global para el Singleton del DataStore
val Context.dataStore by preferencesDataStore(name = "user_settings")

class LocalStorage(private val context: Context) {

    // Definición de las llaves
    private val IS_LOGGED_IN = booleanPreferencesKey("is_logged_in")
    private val USER_EMAIL = stringPreferencesKey("user_email")

    // Guardar sesión y email
    suspend fun saveSession(email: String) {
        context.dataStore.edit { prefs ->
            prefs[IS_LOGGED_IN] = true
            prefs[USER_EMAIL] = email
        }
    }

    // Obtener si está logueado
    suspend fun isLoggedIn(): Boolean {
        val prefs = context.dataStore.data.first()
        return prefs[IS_LOGGED_IN] ?: false
    }

    // Obtener el email guardado
    suspend fun getUserEmail(): String? {
        val prefs = context.dataStore.data.first()
        return prefs[USER_EMAIL]
    }

    // Opcional: Borrar sesión (Logout)
    suspend fun clearSession() {
        context.dataStore.edit { it.clear() }
    }
}

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        // 1. Inicializar dependencias manualmente
        val storage = LocalStorage(applicationContext)
        val loginCase = LoginCase(storage)
        val registerCase = RegisterCase(storage)

        // 2. Estado para controlar qué pantalla mostrar
        // Usamos null como "cargando"
        val startDestination = mutableStateOf<String?>(null)

        lifecycleScope.launch {
            val logged = storage.isLoggedIn()
            startDestination.value = if (logged) "game_screen" else "login_screen"
        }

        setContent {
            val destination = startDestination.value

            if (destination != null) {
                // Aquí usas tu NavHost de Jetpack Compose
                MainNavigation(
                    startDestination = destination,
                    loginCase = loginCase,
                    registerCase = registerCase
                )
            } else {
                // Pantalla de carga mientras lee el DataStore
                Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                    CircularProgressIndicator()
                }
            }
        }
    }
}
¿Cómo funciona el flujo completo?
Arranque: La MainActivity le pregunta al LocalStorage (vía isLoggedIn()) si hay una sesión activa.

Decisión: * Si devuelve true, el startDestination cambia a la pantalla de juego.

Si devuelve false, se queda en la pantalla de login/registro.

Acción del usuario: Cuando el usuario pulsa "Entrar" en tu UI, llamas a loginCase.execute(email, pass).

Persistencia: El UseCase le dice al LocalStorage que guarde el email y ponga el flag en true.

Navegación: Tras el éxito, navegas manualmente a la siguiente pantalla. La próxima vez que el usuario abra la app, el paso 1 detectará que ya está logueado.
