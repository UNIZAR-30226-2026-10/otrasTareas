Para implementar la funcionalidad de selección de fichas, he actualizado las funciones `Tablero` y `FichasStackSnapshot` para recibir los nuevos parámetros y aplicar la lógica de resaltado y clic.

Aquí tienes el código con las modificaciones necesarias:

```kotlin
@Composable
fun Tablero(
    tableroState: TableroSnapshot,
    fichasState: List<FichaSnapshot>,
    seleccionFichas: Boolean = false, // Nuevo parámetro
    seleccionarFicha: (Int) -> Unit = {} // Nuevo parámetro
) {
    Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
        BoxWithConstraints(
            modifier = Modifier
                .aspectRatio(1f)
                .fillMaxWidth()
        ) {
            val tableroSizePx = constraints.maxWidth.toFloat()
            val casillaPx = tableroSizePx / 10

            Card(
                modifier = Modifier
                    .aspectRatio(1f)
                    .fillMaxWidth()
                    .border(2.dp, color_primary, RoundedCornerShape(6.dp)),
                colors = CardDefaults.cardColors(containerColor = color_bg)
            ) {
                Column(modifier = Modifier.fillMaxSize()) {
                    for (fil in 10 downTo 1) {
                        Row(modifier = Modifier.weight(1f)) {
                            for (col in 1..10) {
                                val numCasilla = (fil - 1) * 10 + col
                                val casilla = tableroState.casillas[numCasilla - 1]

                                Box(
                                    modifier = Modifier
                                        .weight(1f)
                                        .fillMaxHeight(),
                                    contentAlignment = Alignment.Center
                                ) {
                                    val casillaImg = when {
                                        casilla.tipo == TipoCasilla.Vacio -> R.drawable.casilla_vacia
                                        casilla.tipo == TipoCasilla.Meta -> R.drawable.casilla_meta
                                        casilla.tipo == TipoCasilla.Bifurcacion -> R.drawable.casilla_bifurcacion
                                        casilla.esCurva -> R.drawable.casilla_curva
                                        casilla.tipo == TipoCasilla.Normal && !casilla.esCurva -> R.drawable.casilla_vertical
                                        else -> R.drawable.casilla_vertical
                                    }

                                    Image(
                                        painter = painterResource(id = casillaImg),
                                        contentDescription = null,
                                        modifier = Modifier
                                            .fillMaxSize()
                                            .rotate(casilla.rotacion.toFloat())
                                    )

                                    Text(
                                        text = numCasilla.toString(),
                                        style = SETextTypes.plano.copy(fontSize = 10.sp),
                                        modifier = Modifier
                                            .align(Alignment.TopStart)
                                            .padding(2.dp)
                                    )

                                    val fichasCasilla = fichasState.filter { it.casilla == numCasilla }
                                    
                                    // Pasamos los nuevos parámetros a la pila de fichas
                                    FichasStackSnapshot(
                                        fichas = fichasCasilla,
                                        seleccionFichas = seleccionFichas,
                                        seleccionarFicha = seleccionarFicha
                                    )
                                }
                            }
                        }
                    }
                }
            }
            ColocarSerpientesEscaleras(tableroState, casillaPx)
        }
    }

@Composable
fun FichasStackSnapshot(
    fichas: List<FichaSnapshot>,
    seleccionFichas: Boolean = false,
    seleccionarFicha: (Int) -> Unit = {}
) {
    val fichasProcesadas: List<Pair<Int, FichaSnapshot>> = if (fichas.size > 2) {
        fichas.groupBy { it.idJugador }.map { it.value.size to it.value.first() }
    } else {
        fichas.map { 1 to it }
    }

    val fractionCasilla = if (fichasProcesadas.size == 1) 0.8f else 0.45f

    FlowColumn(
        modifier = Modifier.fillMaxSize(),
        maxItemsInEachColumn = 2,
        horizontalArrangement = Arrangement.Center,
        verticalArrangement = Arrangement.Center
    ) {
        fichasProcesadas.forEach { (cuenta, ficha) ->
            // Lógica de validación para mostrar recuadro y permitir clic
            val esFichaSeleccionable = seleccionFichas && ficha.esUsuario && !ficha.meta

            Box(
                modifier = Modifier
                    .fillMaxWidth(fractionCasilla)
                    .aspectRatio(1f)
                    // Si es seleccionable, añadimos el borde y el evento de clic
                    .then(
                        if (esFichaSeleccionable) {
                            Modifier
                                .border(2.dp, color_primary, RoundedCornerShape(4.dp))
                                .androidx.compose.foundation.clickable { seleccionarFicha(ficha.id) }
                        } else Modifier
                    ),
                contentAlignment = Alignment.Center
            ) {
                if (ficha.idImg != null) {
                    Box(modifier = Modifier.fillMaxSize(0.9f)) {
                        Image(
                            painter = painterResource(id = ficha.idImg),
                            contentDescription = null,
                            modifier = Modifier.fillMaxSize()
                        )

                        if (cuenta > 1) {
                            Box(
                                modifier = Modifier
                                    .offset(y = (-5).dp, x = 2.dp)
                                    .align(Alignment.TopEnd)
                                    .fillMaxSize(0.5f)
                                    .background(color_bg, CircleShape)
                                    .border(1.dp, color_fg, CircleShape),
                                contentAlignment = Alignment.Center
                            ) {
                                Text(
                                    text = cuenta.toString(),
                                    fontSize = 6.sp,
                                    textAlign = TextAlign.Center
                                )
                            }
                        }
                    }
                }
            }
        }
    }
}
```

### Cambios realizados:
1.  **Parámetros**: He añadido `seleccionFichas: Boolean` y `seleccionarFicha: (Int) -> Unit` tanto a `Tablero` como a `FichasStackSnapshot`.
2.  **Lógica de Selección**: Dentro de `FichasStackSnapshot`, se evalúa la condición `ficha.esUsuario && !ficha.meta` junto con el flag global `seleccionFichas`.
3.  **UI de Selección**: Si la condición se cumple, se aplica un `.border` usando `color_primary` (el amarillo/dorado de tu tema) y se añade el modificador `.clickable` para ejecutar el callback con el `id` de la ficha.
4.  **Selección Única**: Al aplicar la lógica dentro del `forEach` de `fichasProcesadas`, si hay varias fichas en una casilla pero pertenecen a distintos jugadores (o están agrupadas), el sistema permite interactuar de forma individual con la representación visual de esa "pila", cumpliendo con la necesidad de elegir una.
