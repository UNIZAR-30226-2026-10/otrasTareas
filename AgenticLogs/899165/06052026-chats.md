### **POST** /api/lobbies/:lobbyId/invitations
- Envía una invitación a un amigo para que se una al lobby.
- Body: "inviteFrom" y "inviteFor".

### **POST** /api/users/:email/:friendUsername/friends
- Envía al usuario con friendUsername una solicitud de amistad.

### **DELETE** /api/users/:email/friends/:friendUsername
- Elimina a un usuario de la lista de amigos de :email y el usuario correspondiente a :email del amigo borrado.
- Body: "friendUsername".

### **GET** /api/users/:email/friends
- Devuelve la lista de amigos del usuario

import com.UNIZAR_30226_2026_10.SerpientesYEscalerasReMix.domain.repository.AmigosRepository
import com.UNIZAR_30226_2026_10.SerpientesYEscalerasReMix.domain.repository.JugarCrearRepository
import com.UNIZAR_30226_2026_10.SerpientesYEscalerasReMix.domain.repository.LoginRegisterRepository

class AmigosRepositoryImpl(
    private val apiService: ApiService,
    private val loginRepo: LoginRegisterRepository,
    private val jugarCrearRepo: JugarCrearRepository
) : AmigosRepository {

    override suspend fun obtenerAmigos(): List<Usuario> {
        val email = loginRepo.email.value
        if (email.isEmpty()) return emptyList()

        return try {
            val response = apiService.getFriends(email)
            if (response.isSuccessful) {
                response.body()?.map { Usuario(nombre = it.username) } ?: emptyList()
            } else {
                Log.e("AmigosRepo", "Error al obtener amigos: ${response.errorBody()?.string()}")
                emptyList()
            }
        } catch (e: Exception) {
            Log.e("AmigosRepo", "Excepción al obtener amigos", e)
            emptyList()
        }
    }

    override suspend fun anadirAmigo(nombre: String): Boolean {
        val email = loginRepo.email.value
        if (email.isEmpty()) return false

        return try {
            val response = apiService.addFriend(email, nombre)
            response.isSuccessful
        } catch (e: Exception) {
            Log.e("AmigosRepo", "Excepción al añadir amigo", e)
            false
        }
    }

    override suspend fun eliminarAmigo(nombre: String): Boolean {
        val email = loginRepo.email.value
        if (email.isEmpty()) return false

        return try {
            // El DELETE requiere el friendUsername en el body según ApiClient
            val response = apiService.removeFriend(email, nombre, mapOf("friendUsername" to nombre))
            response.isSuccessful
        } catch (e: Exception) {
            Log.e("AmigosRepo", "Excepción al eliminar amigo", e)
            false
        }
    }

    override suspend fun invitarAmigoLobby(nombre: String): Boolean {
        val lobbyId = jugarCrearRepo.lobbyId.value
        val myUsername = loginRepo.username.value
        if (lobbyId.isEmpty() || myUsername.isEmpty()) return false

        return try {
            val response = apiService.sendInvitation(
                lobbyId,
                InvitacionRequest(inviteFrom = myUsername, inviteFor = nombre)
            )
            response.isSuccessful
        } catch (e: Exception) {
            Log.e("AmigosRepo", "Excepción al invitar amigo", e)
            false
        }
    }
}

import kotlinx.coroutines.flow.asStateFlow

class AmigosRepositoryImpl(
    private val apiService: ApiService,
    private val loginRepo: LoginRegisterRepository,
    private val jugarCrearRepo: JugarCrearRepository
) : AmigosRepository {

    private val _amigos = MutableStateFlow<List<Usuario>>(emptyList())
    override val amigos: StateFlow<List<Usuario>> = _amigos.asStateFlow()

    private val _invitaciones = MutableStateFlow<List<Usuario>>(emptyList())
    override val invitaciones: StateFlow<List<Usuario>> = _invitaciones.asStateFlow()

    override suspend fun obtenerAmigos() {
        val email = loginRepo.email.value
        if (email.isEmpty()) return

        try {
            val response = apiService.getFriends(email)
            if (response.isSuccessful) {
                _amigos.value = response.body()?.friends?.map { it.toDomain() } ?: emptyList()
            } else {
                Log.e("AmigosRepo", "Error al obtener amigos: ${response.errorBody()?.string()}")
            }
        } catch (e: Exception) {
            Log.e("AmigosRepo", "Excepción al obtener amigos", e)
        }
    }

    override suspend fun anadirAmigo(nombre: String): Boolean {
        val email = loginRepo.email.value
        if (email.isEmpty()) return false

        return try {
            val response = apiService.addFriend(email, nombre)
            if (response.isSuccessful) {
                obtenerAmigos()
                true
            } else false
        } catch (e: Exception) {
            Log.e("AmigosRepo", "Excepción al añadir amigo", e)
            false
        }
    }

    override suspend fun eliminarAmigo(nombre: String): Boolean {
        val email = loginRepo.email.value
        if (email.isEmpty()) return false

        return try {
            val response = apiService.removeFriend(email, nombre, mapOf("friendUsername" to nombre))
            if (response.isSuccessful) {
                obtenerAmigos()
                true
            } else false
        } catch (e: Exception) {
            Log.e("AmigosRepo", "Excepción al eliminar amigo", e)
            false
        }
    }

    override suspend fun invitarAmigoLobby(nombre: String): Boolean {
        val lobbyId = jugarCrearRepo.lobbyId.value
        val myUsername = loginRepo.username.value
        if (lobbyId.isEmpty() || myUsername.isEmpty()) return false

        return try {
            val response = apiService.sendInvitation(
                lobbyId,
                PostInvitacionRequest(inviteFrom = myUsername, inviteFor = nombre)
            )
            response.isSuccessful
        } catch (e: Exception) {
            Log.e("AmigosRepo", "Excepción al invitar amigo", e)
            false
        }
    }

    override suspend fun obtenerInvitaciones() {
        val email = loginRepo.email.value
        if (email.isEmpty()) return

        try {
            val response = apiService.getInvitations(email)
            if (response.isSuccessful) {
                _invitaciones.value = response.body()?.invites?.map { it.toDomain() } ?: emptyList()
            } else {
                Log.e("AmigosRepo", "Error al obtener invitaciones: ${response.errorBody()?.string()}")
            }
        } catch (e: Exception) {
            Log.e("AmigosRepo", "Excepción al obtener invitaciones", e)
        }
    }

    override suspend fun responderInvitacion(lobbyId: String, inviteFrom: String, aceptar: Boolean): Boolean {
        val myUsername = loginRepo.username.value
        if (myUsername.isEmpty()) return false

        return try {
            val response = apiService.respondInvitation(
                lobbyId,
                AceptarInvitacionRequest(inviteFor = myUsername, inviteFrom = inviteFrom, accept = aceptar)
            )
            if (response.isSuccessful) {
                if (aceptar) {
                    response.body()?.idLobby?.let {
                        jugarCrearRepo.setLobbyId(it)
                    }
                }
                obtenerInvitaciones()
                true
            } else {
                false
            }
        } catch (e: Exception) {
            Log.e("AmigosRepo", "Excepción al responder invitación", e)
            false
        }
    }
