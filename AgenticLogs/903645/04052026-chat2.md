const jsonTablero: SnapshotTableroJSON = tableroAUtilizar.snapshotTableroInicial as SnapshotTableroJSON;
    // Resolver emails de jugadores no-IA: si `idJugador` ya es un email, usarlo; si no, buscar por nombre
    const playerEmailMap: Record<string, string> = {};
    for (const jugador of jugadores.filter(j => !j.esIA)) {
        let email = jugador.idJugador;
        if (!email || !email.includes("@")) {
            const user = await getUserByName(jugador.nombre);
            if (user && (user as Usuario).email) {
                email = (user as Usuario).email;
            } else {
                throw new Error(`No se pudo resolver el email para el jugador ${jugador.nombre}`);
            }
        }
        playerEmailMap[jugador.nombre] = email;
    }

    const partidaCreada = await prisma.partida.create({
        data: {
            estado: Estado.EnCurso,
            snapshotJugadores: jsonJugadores,
            configuracion: {
                tablero,
                numeroJugadores: jugadores.length,
                numeroBots: lobby.numBots
            },
            snapshotTablero: jsonTablero,
            tableroInicialNombre: tablero,

                connect: Object.values(playerEmailMap).map(email => ({ email }))

                        barajaNombre: jugador.nombreMazo!,

            partidaJugadores: true,
